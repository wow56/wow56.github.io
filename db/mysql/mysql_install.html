# Docker 安装

还不会 Docker？Docker 是啥

# 带密码

# 查看可用版本
docker search mysql
# 默认拉取 mysql:latest
docker pull mysql
# MYSQL_ROOT_PASSWORD=123456
# 设置 MySQL 服务 root 用户的密码
docker run -itd --name mysql-test -p 3306:3306 -e MYSQL_ROOT_PASSWORD=123456 mysql

# 正式

# 查看可用版本
docker search mysql
# 默认拉取 mysql:latest
docker pull mysql
# MYSQL_ROOT_PASSWORD=123456
# 设置 MySQL 服务 root 用户的密码
docker run -itd \
    --name mysql-xx \
    -p 3306:3306 \
    -v /home/user/mysql/conf:/etc/mysql/conf.d \
    -v /home/user/mysql/data:/mysql/data \
    -e MYSQL_ROOT_PASSWORD=123456 \
    mysql
